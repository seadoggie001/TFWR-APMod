<Project>
    <PropertyGroup>
        <!-- Path to Steam -->
        <SteamPath>/home/seadoggie/.steam/steam</SteamPath>
        <!-- Path to the game -->
        <GamePath>$(SteamPath)/steamapps/common/The Farmer Was Replaced</GamePath>
        <!-- Path to the game's dlls -->
        <ManagedPath>$(GamePath)/TheFarmerWasReplaced_Data/Managed</ManagedPath>

        <!-- Other paths, probably not used here -->
        <BepInExPath>$(GamePath)/BepInEx</BepInExPath>
        <PluginsPath>$(BepInExPath)/plugins</PluginsPath>
        <MyPluginPath>$(PluginsPath)/$(AssemblyName)</MyPluginPath>
    </PropertyGroup>

    <ItemGroup>
        <!-- These are the internal game files. They allow you to reference internal types and functions --> 
        <Reference Include="$(ManagedPath)/Core.dll">
            <Private>False</Private>
        </Reference>
        <Reference Include="$(ManagedPath)/Utils.dll">
            <Private>False</Private>
        </Reference>

        <!-- These are to reference BepInEx and Unity in code -->
        <Reference Include="BepInEx">
            <HintPath>$(BepInExPath)\core\BepInEx.dll</HintPath>
            <Private>False</Private>
        </Reference>
        <Reference Include="UnityEngine.UI">
            <HintPath>$(ManagedPath)\UnityEngine.UI.dll</HintPath>
            <Private>False</Private>
        </Reference>
        <Reference Include="Unity.TextMeshPro">
            <HintPath>$(ManagedPath)\Unity.TextMeshPro.dll</HintPath>
            <Private>False</Private>
        </Reference>
    </ItemGroup>

    <!-- Copies files to your steam directory after each build -->
    <Target Name="CopyFilesToSteamDirectory" AfterTargets="PostBuildEvent">
        <Message Text="Updating Steam files -> 
            From: $(ProjectDir)$(OutputPath)
            To:   $(MyPluginPath)" Importance="high"  />
        <RemoveDir Directories="$(MyPluginPath)" />
        <MakeDir Directories="$(MyPluginPath)" />
        <ItemGroup>
            <FilesToCopy Include="$(ProjectDir)$(OutputPath)**/*"/>
        </ItemGroup>
        <Copy SourceFiles="@(FilesToCopy)" DestinationFolder="$(MyPluginPath)/%(RecursiveDir)" />
        
        <!-- Use a command like this to auto-start the game after builds -->
<!--        <Exec Command="steam steam://run/2060160" />-->
    </Target>
